# Python program to calculate the series
def calculate_series(num_terms):
    total = 0
    for n in range(num_terms):
        total += (-1)**n / (2*n + 1)
    return total * 4

# Calculate the first 10,000 terms
result = calculate_series(10000)
print(result)
